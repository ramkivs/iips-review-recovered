§7 — Re-deriving digests.json after any intentional edit
    python3 - <<'PY'
    import hashlib, json, os
    files = {}
    for root, _, names in os.walk('.'):
        for name in names:
            p = os.path.relpath(os.path.join(root, name), '.')
            if p in ('README.md', 'digests.json') or p.endswith('.pyc'):
                continue
            b = open(p, 'rb').read()
            files[p] = {'sha256': hashlib.sha256(b).hexdigest(), 'bytes': len(b)}
    doc = json.load(open('digests.json'))
    doc['files'] = dict(sorted(files.items()))
    doc['fileCount'] = len(files)
    open('digests.json', 'w').write(json.dumps(doc, indent=2) + '\n')
    PY
