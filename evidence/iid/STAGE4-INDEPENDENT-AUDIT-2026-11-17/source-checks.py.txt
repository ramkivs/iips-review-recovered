from pathlib import Path
import json,hashlib,math
root=Path.cwd(); bundle=root/'evidence/real-issuer/S4-R3-FILING-REFRESH-2026-11-14'
sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
checks=[]
for name,row in json.loads((bundle/'digests.json').read_text())['files'].items():
 p=bundle/name
 checks.append({'kind':'refresh manifest file','path':str(p.relative_to(root)),'ok':p.exists() and sha(p)==row['sha256'] and p.stat().st_size==row['bytes']})
for name,row in json.loads((bundle/'source/page-map.json').read_text()).items():
 p=bundle/row['file']; origin=root/row['originReport']; txt=origin.read_text(); a=txt.index(row['beginMarker'])+len(row['beginMarker']);b=txt.index(row['endMarker'],a);block=txt[a:b].strip()
 checks.append({'kind':'page excerpt/committed origin identity','name':name,'path':str(p.relative_to(root)),'ok':sha(p)==row['sha256'] and sha(origin)==row['originReportSha256'] and block in p.read_text()})
for row in json.loads((bundle/'runs/04-derived-handchecks.json').read_text())['checks']:
 vals=list(row['inputsCrore'].values());expected=vals[0]-vals[1] if row['formula']=='OCF - capital expenditure' else sum(vals)
 checks.append({'kind':'hand formula arithmetic','name':row['name'],'formula':row['formula'],'reference':'evidence/real-issuer/S4-R3-FILING-REFRESH-2026-11-14/runs/04-derived-handchecks.json','ok':math.isclose(expected,row['resultCrore'],rel_tol=0,abs_tol=1e-8)})

for name,path,terms in anchors:
 lines=(bundle/path).read_text().splitlines();hits=[{'line':i+1} for i,line in enumerate(lines) if any(t.lower() in line.lower() for t in terms)]
