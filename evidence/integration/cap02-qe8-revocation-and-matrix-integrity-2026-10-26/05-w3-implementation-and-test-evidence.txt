  export function resolveActiveBinding(db: DatabaseSync): ActiveOwner {
    db.exec('BEGIN IMMEDIATE;');
    try {
      const owner = resolveActiveBindingInCurrentTransaction(db);
      db.exec('COMMIT;');
      return owner;
    } catch (error) {
      db.exec('ROLLBACK;');
      throw error;
    }
  }

## Consequence for the QE-8 evidence ceiling
  The CAP-02 matrix row QE-8 recorded 'Limitation finding for write/revocation-race (no binding re-read
  inside a write transaction)'. That statement reproduces the Gate 4 act §5 note written before the
  U2 transaction/binding revalidation amendment (IPD evidence/cross-repository/
  IIPS_V3_CAP02_U2_TRANSACTION_BINDING_REVALIDATION_AMENDMENT_2026-10-23.md, EFFECTIVE), which authorized
  resolveActiveBindingInCurrentTransaction and required the request transaction to call it. At the
  designated pin the requirement is implemented. The race limb is therefore T2 component-executable
  (Tier 2 disposable-copy store), not structurally unreachable. It remains component evidence only:
  no full-path PASS is claimable while W-1 refuses every operational write.
